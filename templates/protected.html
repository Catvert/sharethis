{% block extra_scripts %}
const roomId = '{{ room_id }}';

async function unlock() {
    const password = document.getElementById('password').value;
    const error = document.getElementById('error');

    try {
        const res = await fetch(`/api/room/${roomId}/unlock`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ password })
        });

        const data = await res.json();

        if (res.ok && data.success) {
            document.getElementById('unlockForm').style.display = 'none';
            document.getElementById('contentBox').style.display = 'block';
            document.getElementById('content').textContent = data.content;

            if (data.last_view) {
                document.getElementById('warning').style.display = 'block';
                document.getElementById('warning').textContent =
                    'Cette room a été supprimée après cette lecture.';
            } else if (data.views_remaining !== null) {
                document.getElementById('warning').style.display = 'block';
                document.getElementById('warning').textContent =
                    `${data.views_remaining} ouverture(s) restante(s)`;
            }
        } else if (data.expired) {
            error.textContent = 'Cette room a expiré.';
            error.style.display = 'block';
        } else {
            error.textContent = 'Mot de passe incorrect';
            error.style.display = 'block';
        }
    } catch (e) {
        error.textContent = 'Erreur réseau';
        error.style.display = 'block';
    }
}

function copyRoomContent() {
    const content = document.getElementById('content').textContent;
    copyToClipboard(content, 'Contenu copié !');
}

document.getElementById('password').addEventListener('keypress', e => {
    if (e.key === 'Enter') unlock();
});
